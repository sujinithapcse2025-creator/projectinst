def get_conn() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db() -> None:
    with get_conn() as c:
        c.executescript(
            """
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                fullname TEXT NOT NULL,
                dob TEXT,
                mobile TEXT,
                email TEXT,
                language TEXT,
                accessibility TEXT,
                username TEXT UNIQUE NOT NULL,
                password_hash TEXT NOT NULL,
                failed_attempts INTEGER NOT NULL DEFAULT 0,
                locked_until TEXT,
                bank_name TEXT
            );
            CREATE TABLE IF NOT EXISTS guardian (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER NOT NULL,
                guardian_name TEXT,
                relationship TEXT,
                contact TEXT,
                FOREIGN KEY (user_id) REFERENCES users(id)
            );
            CREATE TABLE IF NOT EXISTS verification (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER NOT NULL,
                fingerprint_registered INTEGER DEFAULT 0,
                voice_registered INTEGER DEFAULT 0,
                face_registered INTEGER DEFAULT 0,
                face_image_path TEXT,
                security_question TEXT,
                security_answer_hash TEXT,
                FOREIGN KEY (user_id) REFERENCES users(id)
            );
            """
        )


def hash_text(text: str) -> str:
    return bcrypt.hashpw(text.encode("utf-8"), bcrypt.gensalt()).decode("utf-8")


def check_hash(text: str, hashed: str) -> bool:
    try:
        return bcrypt.checkpw(text.encode("utf-8"), hashed.encode("utf-8"))
    except Exception:
        return False


def normalize_answer(ans: str) -> str:
    return re.sub(r"\s+", " ", ans.strip().lower())


def get_user(username: str):
    with get_conn() as c:
        return c.execute("SELECT * FROM users WHERE username = ?", (username.strip(),)).fetchone()


def get_verification(user_id: int):
    with get_conn() as c:
        return c.execute("SELECT * FROM verification WHERE user_id = ?", (user_id,)).fetchone()


def is_locked(user) -> bool:
    if not user or not user["locked_until"]:
        return False
    try:
        return dt.datetime.fromisoformat(user["locked_until"]) > dt.datetime.now()
    except ValueError:
        return False


def set_failed_attempts(user_id: int, n: int) -> None:
    with get_conn() as c:
        c.execute("UPDATE users SET failed_attempts = ? WHERE id = ?", (n, user_id))


def lock_user(user_id: int) -> None:
    until = (dt.datetime.now() + dt.timedelta(minutes=LOCK_MINUTES)).isoformat()
    with get_conn() as c:
        c.execute("UPDATE users SET locked_until = ? WHERE id = ?", (until, user_id))

# --------------------------------------------------------------------------
# TEXT TO SPEECH  (pyttsx3 first, browser speech as fallback)
# --------------------------------------------------------------------------
def _pyttsx_say(text: str) -> None:
    try:
        engine = pyttsx3.init()
        engine.setProperty("rate", 150)
        engine.say(text)
        engine.runAndWait()
    except Exception:
        pass


def _browser_say(text: str, lang_code: str) -> None:
    components.html(
        f"""<script>
        const u = new SpeechSynthesisUtterance({json.dumps(text)});
        u.lang = {json.dumps(lang_code)}; u.rate = 0.9;
        window.speechSynthesis.cancel(); window.speechSynthesis.speak(u);
        </script>""",
        height=0,
    )


def speak(text: str, force: bool = False) -> None:
    """Reusable text-to-speech helper."""
    if not (st.session_state.voice_on or force):
        return
    code = LANG_CODE_BY_KEY.get(st.session_state.lang_key, "en-IN")
    if pyttsx3 and code == "en-IN":
        threading.Thread(target=_pyttsx_say, args=(text,), daemon=True).start()
    else:
        _browser_say(text, code)


def read_aloud_button(text: str, key: str) -> None:
    if st.button(t("read_page"), key=key):
        speak(text, force=True)

def banner() -> None:
    st.markdown(f'<div class="abc-banner" role="alert">⚠️ {PROTOTYPE_WARNING}</div>', unsafe_allow_html=True)


def card(msg: str, kind: str = "success", icon: str = "") -> None:
    st.markdown(f'<div class="abc-card abc-{kind}" role="status">{icon} {msg}</div>', unsafe_allow_html=True)


def feedback(msg: str, kind: str, speak_it: bool = True) -> None:
    """Show a message as a visible caption card AND read it aloud."""
    icon = {"success": "✅", "warning": "⚠️", "error": "❌"}[kind]
    card(msg, kind, icon)
    if speak_it:
        speak(msg)


def step_progress(step: int, total: int = 4) -> None:
    st.progress(int(step / total * 100), text=f"Step {step} of {total} — {int(step / total * 100)}%")


def flash_show() -> None:
    f = st.session_state.flash
    if f:
        feedback(f["msg"], f["kind"])
        st.session_state.flash = None

def password_problems(pw: str) -> list:
    issues = []
    if len(pw) < 8:
        issues.append("at least 8 characters")
    if not re.search(r"[A-Z]", pw):
        issues.append("one uppercase letter")
    if not re.search(r"[a-z]", pw):
        issues.append("one lowercase letter")
    if not re.search(r"\d", pw):
        issues.append("one number")
    if not re.search(r"[^A-Za-z0-9]", pw):
        issues.append("one special character")
    return issues

def page_accessibility():
    st.header(t("access_title"))
    speak(t("access_title"))
    choice = st.multiselect("Select all that apply", ACCESS_OPTIONS, default=st.session_state.access)
    c1, c2 = st.columns(2)
    if c1.button(t("back"), key="acc_back"):
        go("language")
    if c2.button(t("continue"), key="acc_next"):
        if not choice:
            feedback("Please choose at least one option, or 'Prefer not to disclose'.", "warning")
            return
        st.session_state.access = choice
        # Customize the interface from the selection
        if ACCESS_OPTIONS[0] in choice:
            st.session_state.voice_on = True
            st.session_state.high_contrast = True
            st.session_state.font_px = max(st.session_state.font_px, 22)
        if ACCESS_OPTIONS[4] in choice or ACCESS_OPTIONS[2] in choice:
            st.session_state.font_px = max(st.session_state.font_px, 24)
        go("verify")

def page_verify():
    ver, step = st.session_state.ver, st.session_state.ver_step
    st.header("Four-Step Identity Verification")
    step_progress(min(step, 4) if step <= 4 else 4)

    if step == 1:
        st.subheader("Step 1: Fingerprint Registration")
        st.caption("Simulated biometric registration for the demo.")
        if st.button("Register Fingerprint", key="fp_reg"):
            with st.spinner("Scanning fingerprint..."):
                time.sleep(1.2)
            ver["fp"] = True
        if ver["fp"]:
            feedback("Fingerprint Registered", "success")
            if st.button(t("continue"), key="fp_next"):
                st.session_state.ver_step = 2
                st.rerun()

    elif step == 2:
        st.subheader("Step 2: Voice Registration")
        st.info('Please say: "My voice is my secure banking identity."')
        recorded = None
        if hasattr(st, "audio_input"):
            recorded = st.audio_input("Start Recording")
        c1, c2 = st.columns(2)
        if c1.button("Save Voice", key="voice_save"):
            if recorded is not None or st.session_state.get("voice_sim"):
                with st.spinner("Enrolling voice..."):
                    time.sleep(1)
                ver["voice"] = True
            else:
                feedback("Please record your voice first, or use the demo option.", "warning")

    with st.form("register_form"):

    if not submitted:
        return

    errors = []
    if len(fullname.strip()) < 3:
        errors.append("Please enter your full name.")
    if not re.fullmatch(r"[6-9]\d{9}", mobile.strip()):
        errors.append("Enter a valid 10-digit Indian mobile number.")
    if not re.fullmatch(r"[^@\s]+@[^@\s]+\.[^@\s]+", email.strip()):
        errors.append("Enter a valid email address.")
    if not re.fullmatch(r"[A-Za-z0-9_.]{4,20}", username.strip()):
        errors.append("Username must be 4-20 letters, numbers, dot or underscore.")
    if (p := password_problems(pw)):
        errors.append("Password needs: " + ", ".join(p) + ".")
    if pw != pw2:
        errors.append("Passwords do not match.")
    if not errors and get_user(username):
        errors.append("That username is already taken.")

    if errors:
        for e in errors:
            card(e, "error", "❌")
        speak(errors[0])
        return

    try:
        with get_conn() as c:
            cur = c.execute(
                """INSERT INTO users (fullname, dob, mobile, email, language, accessibility,
                                      username, password_hash)
                   VALUES (?,?,?,?,?,?,?,?)""",

                 st.session_state.lang_name, ", ".join(st.session_state.access),

                (uid, int(ver["fp"]), int(ver["voice"]), int(ver["face"]), ver["face_path"],
                 ver["question"], ver["answer_hash"]),
            )
    except sqlite3.IntegrityError:
        feedback("That username is already taken.", "error")
        return

    st.session_state.flash = {"msg": "Registration successful. Please login.", "kind": "success"}
    go("login")

def handle_login(username: str, password: str) -> None:
    if not username or not password:
        feedback("Please enter both username and password.", "warning")
        return

    user = get_user(username)

    if user and is_locked(user):
        feedback(t("locked"), "error")
        return

    if user and check_hash(password, user["password_hash"]):
        # SUCCESS: reset failed attempts, log in, go to dashboard
        set_failed_attempts(user["id"], 0)
        st.session_state.logged_in = True
        st.session_state.user_id = user["id"]
        st.session_state.username = user["username"]
        st.session_state.anon_fail = 0
        st.session_state.last_active = time.time()
        go("dashboard" if user["bank_name"] else "bank")
        return

    # FAILURE: generic message, never reveal which field was wrong
    if user:
        fails = user["failed_attempts"] + 1
        set_failed_attempts(user["id"], fails)
    else:
        st.session_state.anon_fail += 1
        fails = st.session_state.anon_fail

    remaining = max(0, MAX_LOGIN_ATTEMPTS - fails)
    msg = t("wrong")
    feedback(msg, "error")                       # shown as text caption + read aloud
    st.caption(f"{t('attempts_left')}: {remaining}")

    if fails >= MAX_LOGIN_ATTEMPTS:              # -> Stage 2 automatically
        st.session_state.recovery_user = username
        st.session_state.kba_fail = 0
        time.sleep(1.2)
        go("recovery_kba")

def page_recovery_kba():
    username = st.session_state.recovery_user
    if not username:
        go("login")
        return

    st.header(t("kba_title"))
    st.write(t("kba_intro"))
    step_progress(1, 3)

    user = get_user(username)
    if user and is_locked(user):
        feedback(t("locked"), "error")
        if st.button(t("back"), key="kba_locked_back"):
            reset_recovery_state()
            go("login")
        return

    ver = get_verification(user["id"]) if user else None
    question = ver["security_question"] if ver and ver["security_question"] else SECURITY_QUESTIONS[2]
    st.markdown(f"**{question}**")
    speak(question)

    with st.form("kba_form"):
        ans = st.text_input(t("kba_answer"), type="password")
        submit = st.form_submit_button(t("verify_btn"))

    if submit:
        correct = bool(user and ver and ver["security_answer_hash"]
                       and check_hash(normalize_answer(ans), ver["security_answer_hash"]))
        if correct:
            st.session_state.kba_ok = True
            st.session_state.re_fp = st.session_state.re_voice = st.session_state.re_face = False
            go("recovery_reverify")
        else:
            st.session_state.kba_fail += 1
            left = MAX_KBA_ATTEMPTS - st.session_state.kba_fail
            if left <= 0:
                if user:
                    lock_user(user["id"])
                feedback(t("locked"), "error")
                reset_recovery_state()
                time.sleep(1.5)
                go("login")
            else:
                feedback(f"{t('kba_wrong')} {t('attempts_left')}: {left}", "error")

    if st.button(t("back"), key="kba_back"):
        reset_recovery_state()
        go("login")

    st.caption("Use 8+ characters with upper & lower case letters, a number and a special character.")

    with st.form("reset_form"):
        new_pw = st.text_input(t("new_pw"), type="password")
        confirm = st.text_input(t("confirm_pw"), type="password")
        submit = st.form_submit_button(t("reset_btn"))

    if submit:

        if problems:
            feedback("Password needs: " + ", ".join(problems) + ".", "error")

            feedback("Passwords do not match.", "error")

            st.session_state.flash = {"msg": t("reset_ok"), "kind": "success"}

# ---- Voice assistant ----
def interpret(cmd: str):
    c = cmd.lower().strip()
    if any(k in c for k in ["balance", "இருப்பு", "बैलेंस", "शेष"]):
        return f"Your available balance is {SAMPLE_BALANCE:,.0f} rupees.", None
    if any(k in c for k in ["transaction", "பரிவர்த்தனை", "लेन"]):
        return tx_sentences(), None
    if "bill" in c:
        return "Opening bill payments. This is a prototype, so no real payment will be made.", None
    if any(k in c for k in ["safety", "tip", "fraud", "பாதுகாப்பு", "सुरक्षा"]):
        return "Never share your OTP, PIN or password. Your bank will never ask for them.", "fraud"
    if any(k in c for k in ["dashboard", "home", "முகப்பு", "डैशबोर्ड"]):
        return "Opening your dashboard.", "dashboard"
    return "Sorry, I did not understand. Try: Check Balance, Recent Transactions, Open Dashboard or Safety Tips.", None


def transcribe(audio_file) -> str | None:
    if sr is None:
        return None
    recognizer = sr.Recognizer()
    try:
        with sr.AudioFile(audio_file) as source:
            audio = recognizer.record(source)
        return recognizer.recognize_google(audio, language=LANG_CODE_BY_KEY.get(st.session_state.lang_key, "en-IN"))
    except Exception:
        return None

}

SIDEBAR_PAGES = {"Dashboard": "dashboard", "Voice Assistant": "voice",
                 "Fraud Center": "fraud", "Settings": "settings"}


def main() -> None:
    init_db()
    init_state()
    inject_css()

    # Session timeout
    if st.session_state.logged_in and time.time() - st.session_state.last_active > SESSION_TIMEOUT_MIN * 60:
        for k in ("logged_in", "user_id", "username"):
            st.session_state[k] = DEFAULTS[k]
        st.session_state.flash = {"msg": "Session expired for your safety. Please login again.", "kind": "warning"}
        st.session_state.page = "login"
    st.session_state.last_active = time.time()

    banner()

    if st.session_state.logged_in:
        with st.sidebar:
            st.markdown(f"**{st.session_state.username}**")
            for label, pg in SIDEBAR_PAGES.items():
                if st.button(label, key=f"nav_{pg}"):
                    go(pg)
            if st.button(t("logout"), key="nav_logout"):
                logout()

    PAGES.get(st.session_state.page, page_welcome)()

    st.divider()
    st.caption("INNOVEX · Dexathon 2026 · " + PROTOTYPE_WARNING)


if __name__ == "__main__":
    main()
